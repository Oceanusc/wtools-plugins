// This document is always loaded in an opaque-origin sandbox without host APIs.
const url = URL.createObjectURL(new Blob([__WTOOLS_FILTER_WORKER__], { type: 'text/javascript' }));
const worker = new Worker(url);
URL.revokeObjectURL(url);
worker.onmessage = event => {
  if (!event.data?.ready) worker.terminate();
  parent.postMessage(event.data, '*');
};
worker.onerror = () => { worker.terminate(); parent.postMessage({ ok: false, error: '过滤执行失败' }, '*'); };
addEventListener('message', event => { if (event.source === parent) worker.postMessage(event.data); });
